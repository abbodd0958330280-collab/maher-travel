/* ========================================= MAHER TRAVEL Main Application Script ========================================= */ const STORAGE_KEY = “maherTravelBookings”; /* —————————————– Helpers —————————————– */ function getBookings() { try { return JSON.parse(localStorage.getItem(STORAGE_KEY)) || []; } catch (error) { console.error(“Could not read bookings:”, error); return []; } } function saveBookings(bookings) { localStorage.setItem(STORAGE_KEY, JSON.stringify(bookings)); } function generateReference() { const random = Math.floor(100000 + Math.random() * 900000); return MH-${random}; } function showTab(tabId) { document.querySelectorAll(”.tab-content”).forEach(section => { section.classList.remove(“active”); }); const selected = document.getElementById(tabId); if (selected) { selected.classList.add("active"); } document.querySelectorAll(".nav-btn").forEach(button => { button.classList.remove("active"); if (button.dataset.tab === tabId) { button.classList.add("active"); } }); window.scrollTo({ top: 0, behavior: "smooth" }); } /* —————————————– Navigation —————————————– */ document.addEventListener(“DOMContentLoaded”, () => { document.querySelectorAll(".nav-btn").forEach(button => { button.addEventListener("click", () => { showTab(button.dataset.tab); }); }); setupBookingForm(); setMinimumDate(); }); /* —————————————– Minimum booking date —————————————– */ function setMinimumDate() { const dateInput = document.getElementById("travelDate"); if (!dateInput) return; const today = new Date(); const year = today.getFullYear(); const month = String(today.getMonth() + 1).padStart(2, "0"); const day = String(today.getDate()).padStart(2, "0"); dateInput.min = `${year}-${month}-${day}`; } /* —————————————– Booking Form —————————————– */ function setupBookingForm() { const form = document.getElementById("bookingForm"); if (!form) return; form.addEventListener("submit", function(event) { event.preventDefault(); const booking = { id: generateReference(), name: document.getElementById("customerName").value.trim(), phone: document.getElementById("customerPhone").value.trim(), pickup: document.getElementById("pickup").value.trim(), destination: document.getElementById("destination").value.trim(), date: document.getElementById("travelDate").value, time: document.getElementById("travelTime").value, passengers: document.getElementById("passengers").value, vehicle: document.getElementById("vehicle").value, notes: document.getElementById("notes").value.trim(), status: "new", createdAt: new Date().toISOString() }; /* Validate */ if ( !booking.name || !booking.phone || !booking.pickup || !booking.destination || !booking.date || !booking.time || !booking.passengers || !booking.vehicle ) { alert("يرجى تعبئة جميع الحقول المطلوبة."); return; } /* Save */ const bookings = getBookings(); bookings.push(booking); saveBookings(bookings); /* Show confirmation */ const referenceElement = document.getElementById("referenceNumber"); if (referenceElement) { referenceElement.textContent = booking.id; } const modal = document.getElementById("confirmationModal"); if (modal) { modal.style.display = "flex"; } form.reset(); }); } /* —————————————– Track Booking —————————————– */ function trackBooking() { const input = document.getElementById("trackingNumber"); const result = document.getElementById("statusResult"); if (!input || !result) return; const reference = input.value.trim().toUpperCase(); if (!reference) { result.innerHTML = `
⚠️

أدخل رقم الحجز أولاً.
`; return; } const bookings = getBookings(); const booking = bookings.find(item => item.id === reference); if (!booking) { result.innerHTML = `
❌

لم يتم العثور على هذا الحجز.
تأكد من رقم الحجز وحاول مرة أخرى.
`; return; } const statusText = { new: "طلب جديد", confirmed: "تم تأكيد الحجز", rejected: "تم رفض الحجز", cancelled: "تم إلغاء الحجز" }; const statusClass = booking.status || "new"; result.innerHTML = `

${escapeHTML(booking.id)}

${statusText[statusClass] || "قيد المراجعة"}
المسافر ${escapeHTML(booking.name)}
من ${escapeHTML(booking.pickup)}
إلى ${escapeHTML(booking.destination)}
التاريخ ${escapeHTML(booking.date)}
الوقت ${escapeHTML(booking.time)}
عدد الركاب ${escapeHTML(booking.passengers)}
`; } /* —————————————– Modal —————————————– */ function closeModal() { const modal = document.getElementById("confirmationModal"); if (modal) { modal.style.display = "none"; } showTab("tracking"); } function copyReference() { const reference = document.getElementById("referenceNumber"); if (!reference) return; const text = reference.textContent.trim(); if (navigator.clipboard) { navigator.clipboard.writeText(text) .then(() => { alert("تم نسخ رقم الحجز."); }) .catch(() => { fallbackCopy(text); }); } else { fallbackCopy(text); } } function fallbackCopy(text) { const textarea = document.createElement("textarea"); textarea.value = text; document.body.appendChild(textarea); textarea.select(); try { document.execCommand("copy"); alert("تم نسخ رقم الحجز."); } catch (error) { alert(`رقم الحجز: ${text}`); } textarea.remove(); } /* —————————————– Security helper —————————————– */ function escapeHTML(value) { if (value === null || value === undefined) { return ""; } return String(value) .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); } /* —————————————– Close modal when clicking outside —————————————– */ window.addEventListener(“click”, function(event) { const modal = document.getElementById("confirmationModal"); if ( modal && event.target === modal ) { modal.style.display = "none"; } }); /* —————————————– Expose functions —————————————– */ window.showTab = showTab; window.trackBooking = trackBooking; window.closeModal = closeModal; window.copyReference = copyReference;
